! ============================================================
! Berlin1 — Edge Router 1
! OSPF Router-ID 1.1.1.1, NAT, Telephony (CME), ACLs
! Berlin Campus Network
! ============================================================

enable
configure terminal

hostname Berlin1
no ip domain-lookup

line console 0
 logging synchronous
line vty 0 15
 logging synchronous
exit
banner motd #Authorised access only#

! --- WAN Links to ISPs ---
interface s0/0/0
 ip address 208.0.0.2 255.255.255.252
 ip nat outside
 no shutdown

interface s0/0/1
 ip address 208.0.0.10 255.255.255.252
 ip nat outside
 no shutdown

! --- LAN Links to Distribution Switches ---
! Connection to B.MLS1
interface f0/0
 ip address 172.16.1.2 255.255.255.252
 ip nat inside
 no shutdown

! Connection to B.MLS2
interface f0/1
 ip address 172.16.4.2 255.255.255.252
 ip nat inside
 no shutdown

! --- Default Routes (primary via ISP1, backup via ISP2) ---
ip route 0.0.0.0 0.0.0.0 Serial0/0/0
ip route 0.0.0.0 0.0.0.0 Serial0/0/1 3

! --- OSPF ---
router ospf 1
 router-id 1.1.1.1
 log-adjacency-changes
 passive-interface Serial0/0/0
 passive-interface Serial0/0/1
 network 172.16.1.0 0.0.0.3 area 0
 network 172.16.4.0 0.0.0.3 area 0
 default-information originate

! --- NAT (PAT Overload) ---
access-list 1 permit 172.16.0.0 0.0.255.255
access-list 2 permit 172.16.0.0 0.0.255.255
ip nat inside source list 1 interface Serial0/0/0 overload
ip nat inside source list 2 interface Serial0/0/1 overload
